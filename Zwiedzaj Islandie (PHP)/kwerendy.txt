1. SELECT idObiekt, plik, nazwa FROM `obiekty` WHERE panstwo = "Islandia";

2. SELECT plik, nazwa, nazwaCechy, wartoscCechy, opis, rodzaj FROM `obiekty` JOIN `rodzaje` ON obiekty.idRodzaj = rodzaje.idRodzaj WHERE idObiekt = 46;

3. SELECT nazwa FROM `obiekty` WHERE panstwo = "Islandia" AND idRodzaj = 10;

4. SELECT rodzaj, COUNT(nazwa) AS "Liczba obiektów" FROM `rodzaje` JOIN `obiekty` ON rodzaje.idRodzaj = obiekty.idRodzaj GROUP BY nazwa;
